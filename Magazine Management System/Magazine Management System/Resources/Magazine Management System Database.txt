--creating tables
create TABLE userRoles (
Id INT primary key,
Name VARCHAR(100) NOT NULL 
);

create TABLE Users (
id int PRIMARY KEY,
Name VARCHAR2(100) NOT NULL,
Email VARCHAR2(100) NOT NULL ,
CONSTRAINT chk_email CHECK (Email LIKE '%@%.%'),
Password VARCHAR2(100) NOT NULL,
Role_ID NUMBER(6) NOT NULL,
PhoneNumber VARCHAR2(11) NOT NULL,
CONSTRAINT chk_phone_length CHECK (LENGTH(PhoneNumber) = 11),
SecQues NUMBER(3) UNIQUE NOT NULL,
ProfilePic varchar2(200) NOT NULL,
CONSTRAINT fk_role
        FOREIGN KEY (Role_ID)
        REFERENCES userRoles(Id)
        ON DELETE CASCADE
);

CREATE TABLE Sections (
    ID NUMBER PRIMARY KEY NOT NULL,
    name VARCHAR2(255),
    description VARCHAR2(600)
);

create table Articles (
Id Number(5) primary key,
Title  varchar2(200) NOT NULL,
Content  varchar2(4000) NOT NULL,
Publisher_Name   varchar2(100) NOT NULL,
Section_Id Number(5),
Photo varchar2(200) NOT NULL,
Published_Time  DATE NOT NULL,
FOREIGN KEY (Section_Id) REFERENCES Sections(ID)ON DELETE CASCADE);

CREATE TABLE Favorites (
    user_id    NUMBER,
    article_id NUMBER,
    
    CONSTRAINT pk_favorites PRIMARY KEY (user_id, article_id),
    
    CONSTRAINT fk_users FOREIGN KEY (user_id) REFERENCES Users(ID)ON DELETE CASCADE,
    CONSTRAINT fk_article FOREIGN KEY (article_id) REFERENCES Articles(ID)ON DELETE CASCADE
);

CREATE TABLE Followed_Sections (
    user_id     NUMBER,
    section_id  NUMBER,
    
    CONSTRAINT pk_followed_sections
        PRIMARY KEY (user_id, section_id),
    
    CONSTRAINT fk_follow_user
        FOREIGN KEY (user_id)
        REFERENCES Users(ID)
        ON DELETE CASCADE,

    CONSTRAINT fk_follow_section
        FOREIGN KEY (section_id)
        REFERENCES Sections(ID)
        ON DELETE CASCADE
);

CREATE TABLE Reactions (
    user_id     NUMBER,
    article_id  NUMBER,

    CONSTRAINT pk_reactions
        PRIMARY KEY (user_id, article_id),

    CONSTRAINT fk_react_user
        FOREIGN KEY (user_id)
        REFERENCES Users(ID)
        ON DELETE CASCADE,

    CONSTRAINT fk_react_article
        FOREIGN KEY (article_id)
        REFERENCES Articles(ID)
        ON DELETE CASCADE
);

CREATE TABLE Comments (
Id          NUMBER PRIMARY KEY,
Article_id  NUMBER,
User_id     NUMBER,
Comment_Time DATE,
Commenttext    VARCHAR2(2000),

CONSTRAINT fk_articles FOREIGN KEY (Article_id) REFERENCES Articles(Id)ON DELETE CASCADE,
CONSTRAINT fk_userss    FOREIGN KEY (User_id) REFERENCES Users(Id)ON DELETE CASCADE
);

create table Notifications(
NotificationId int primary key,
UserId int,
ArticleId number,
Notification_content varchar(1000) not null,
NotificationTime date,
IsRead  NUMBER(1) DEFAULT 0,

constraint fk_user foreign key(UserId) references Users(Id)ON DELETE CASCADE,
constraint fk_artiNoti foreign key(ArticleId) references Articles(Id)ON DELETE CASCADE
);

BEFORE INSERT ON Users

    IF :NEW.id IS NULL THEN

-- SearchArticles
CREATE OR REPLACE PROCEDURE SearchArticles (
    search_value IN VARCHAR2,
    out_cursor OUT SYS_REFCURSOR
) AS
    search_date DATE;
BEGIN
    BEGIN
        search_date := TO_DATE(search_value, 'YYYY-MM-DD');
        
        OPEN out_cursor FOR
            SELECT 
                A.Id,
                A.Title,
                A.Photo AS ImagePath,
                A.Published_Time AS PublishDate,
                (SELECT COUNT(*) FROM Comments C WHERE C.Article_id = A.Id) AS CommentCount,
                (SELECT COUNT(*) FROM Reactions R WHERE R.article_id = A.Id) AS Likes
            FROM Articles A
            WHERE TRUNC(A.Published_Time) = search_date
            ORDER BY A.Published_Time DESC;
    
    EXCEPTION
        WHEN OTHERS THEN
            OPEN out_cursor FOR
                SELECT 
                    A.Id,
                    A.Title,
                    A.Photo AS ImagePath,
                    A.Published_Time AS PublishDate,
                    (SELECT COUNT(*) FROM Comments C WHERE C.Article_id = A.Id) AS CommentCount,
                    (SELECT COUNT(*) FROM Reactions R WHERE R.article_id = A.Id) AS Likes

                FROM Articles A
                WHERE LOWER(A.Title) LIKE '%' || LOWER(search_value) || '%'
                   OR LOWER(A.Publisher_Name) LIKE '%' || LOWER(search_value) || '%'
                ORDER BY A.Published_Time DESC;
    END;
END;
/

-- GetLatestArticles
CREATE OR REPLACE PROCEDURE GetLatestArticles (
    result_cursor OUT SYS_REFCURSOR
)
IS
BEGIN
    OPEN result_cursor FOR
        SELECT 
            A.Id,
            A.Title,
            A.Photo AS ImagePath,
            A.Published_Time AS PublishDate,
            (SELECT COUNT(*) FROM Comments C WHERE C.Article_id = A.Id) AS CommentCount,
            (SELECT COUNT(*) FROM Reactions R WHERE R.article_id = A.Id) AS Likes
        FROM Articles A
        ORDER BY A.Published_Time DESC;
END;

-- Get_Section_Id 
CREATE OR REPLACE PROCEDURE Get_Section_Id (
    section_name IN VARCHAR2,
    section_id OUT NUMBER
)
AS
BEGIN
    SELECT id INTO section_id
    FROM Sections
    WHERE name = section_name;

EXCEPTION
    WHEN NO_DATA_FOUND THEN
        section_id := -1;
    WHEN OTHERS THEN
        section_id := -2; -- Other Exception
END;

-- GetNoti
CREATE OR REPLACE PROCEDURE GetNoti (
    uderId IN NUMBER,
    NotificationInfo OUT SYS_REFCURSOR
)
AS
BEGIN
OPEN NotificationInfo FOR
SELECT n.Notification_Content,a.Title,s.Name AS SectionName,n.NotificationTime,
n.IsRead,n.NOTIFICATIONID,n.ARTICLEID
FROM Notifications n JOIN Articles a ON a.Id = n.ArticleId JOIN Sections s ON a.Section_Id = s.Id
WHERE n.UserId = uderId
        ORDER BY 
            n.NotificationTime DESC;    
END;
